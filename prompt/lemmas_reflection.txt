```isabelle
theorem min_priority_leq_all:
assumes a1: "¬is_empty q" and a2: "v ∈ set |q|"
shows "the (priority q (min q)) ≤ the (priority q v)"
proof -
(* Step 1: Unfold internal representation and obtain non‑empty list *)
let ?L = "alist_of q"
have L_nonempty: "?L ≠ []"
using is_empty_iff_alist_eq_nil  by auto    (*@>>> lemma <<< *)
(* Step 2: Express the minimum element and its priority *)
obtain m p_min where hd_eq: "(m, p_min) = hd ?L"
by (metis surj_pair)
have min_eq: "min q = m"
unfolding min_def hd_eq
  by (metis fst_eqD hd_eq) 
(* Head belongs to the list *)
have head_in_set: "(m, p_min) ∈ set ?L"
using L_nonempty
  by (simp add: hd_eq) 
(* Distinct keys imply map_of gives the unique value *)
have distinct_fst: "distinct (map fst ?L)"
using alist_of_distinct[of q] .
have priority_min_some: "priority q (min q) = Some p_min"
unfolding priority_def min_eq
using map_of_member_if_key_in_set_and_distinct[OF distinct_fst head_in_set] by simp
(* Step 3: Characterise the arbitrary value v *)
from a2 have v_in_map_fst: "v ∈ set (map fst ?L)"
unfolding values_def .
then obtain p_v where pair_in_set: "(v, p_v) ∈ set ?L"
using values_mem_implies_exists_pair[OF a2] by blast
have priority_v_some: "priority q v = Some p_v"
unfolding priority_def
using map_of_member_if_key_in_set_and_distinct[OF distinct_fst pair_in_set] by simp
(* Step 4: Exploit sortedness invariant to get inequality *)
have sorted_snd: "sorted (map snd ?L)"
using alist_of_sorted[of q] .
have min_priority_le_all: "∀x ∈ set ?L. snd (hd ?L) ≤ snd x"
using sorted_proj_hd_min[OF sorted_snd L_nonempty] by simp
hence "p_min ≤ p_v"
using hd_eq pair_in_set
  by (metis sndI) 
(* Step 5: Translate Some-expressions using the *)
have the_min: "the (priority q (min q)) = p_min"
using priority_min_some by simp
have the_v: "the (priority q v) = p_v"
using priority_v_some by simp
(* Step 6: Conclude *)
show ?thesis
using the_min the_v ‹p_min ≤ p_v› by auto
qed
```
