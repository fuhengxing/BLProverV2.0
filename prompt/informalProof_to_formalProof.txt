 assumes a1 :"¬is_empty q " and a2:"v ∈ set|q|"

proof -  
  (* Step 1: Unfold the internal representation and confirm the list is non-empty *)  
  let ?L = "alist_of q"  
  have h1: "?L ≠ []"  
    using a1 is_empty_iff_alist_eq_nil by auto
  (* Step 2: Define the minimum element and its priority, expressed as hd ?L *)  
  obtain m p_min where hd_def: "(m, p_min) = hd ?L"
    by auto
  have h2: "min q = m"  
    unfolding min_def hd_def by simp
  have h3: "(m, p_min) ∈ set ?L"  
    using h1 hd_in_set_if_nonempty by auto
  have h4: "priority q (min q) = Some p_min"  
  proof -  
    have h4a: "distinct (map fst ?L)"  
      using pri_queue_distinct_fst[of q] .  
    with h3 show ?thesis  
      unfolding priority_def h2 hd_def
      by simp

  have h5: "v ∈ set (map fst ?L)"

      unfolding priority_def
      by auto
